cmake_minimum_required(VERSION 3.16)
project(JHY_Mecanum_Simulation C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(mecanum_sim src/main.c src/config.c src/sim.c src/editor.c src/tests.c src/io.c
    ../JHY/Src/alg_mecanum_wheel_chassis.c)
target_include_directories(mecanum_sim PRIVATE src ../JHY/Inc)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(mecanum_sim PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(NOT WIN32)
        target_link_libraries(mecanum_sim PRIVATE m)
    endif()
endif()
if(WIN32)
    target_link_libraries(mecanum_sim PRIVATE shell32)
endif()
enable_testing()
add_test(NAME core_models_and_safety COMMAND mecanum_sim --self-test)
